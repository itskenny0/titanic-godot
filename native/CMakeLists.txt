cmake_minimum_required(VERSION 3.16)
project(titanic_runtime C)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(../vendor/quickjs quickjs EXCLUDE_FROM_ALL)
add_library(titanic SHARED bridge.c)
target_include_directories(titanic PRIVATE ../vendor/godot-headers ../vendor/quickjs)
target_link_libraries(titanic PRIVATE qjs)
set(TITANIC_GO_LIBRARY "" CACHE FILEPATH "Go codec archive or shared library")
if(TITANIC_GO_LIBRARY)
 target_compile_definitions(titanic PRIVATE TITANIC_GO_CODECS)
 target_link_libraries(titanic PRIVATE "${TITANIC_GO_LIBRARY}")
 if(UNIX AND NOT APPLE)
  target_link_libraries(titanic PRIVATE pthread dl m)
 endif()
endif()
set_target_properties(titanic PROPERTIES PREFIX "lib")
if(MINGW)
 target_link_options(titanic PRIVATE -static-libgcc)
endif()
